Step 3: list and export every image, then find the frame

Save as list_textures.py and run python list_textures.py SNEAKARCADE-STATIC-PROD-v12-1024-NoComp.data out/:

import sys, os, json, UnityPy
src, out = sys.argv[1], sys.argv[2]
os.makedirs(out, exist_ok=True)
env = UnityPy.load(src)
rows = []
for obj in env.objects:
    if obj.type.name not in ("Texture2D", "Sprite"):
        continue
    try:
        d = obj.read()
        name = getattr(d, "m_Name", None) or getattr(d, "name", "") or f"id{obj.path_id}"
        img = d.image
        w, h = img.size
        fn = f"{obj.type.name}_{obj.path_id}_{name}".replace("/", "_")[:120] + ".png"
        img.save(os.path.join(out, fn))
        rows.append({"type": obj.type.name, "path_id": obj.path_id, "name": name, "w": w, "h": h,
                     "file": fn, "container": obj.container if hasattr(obj, "container") else None})
    except Exception as e:
        rows.append({"type": obj.type.name, "path_id": obj.path_id, "error": str(e)})
rows.sort(key=lambda r: -(r.get("w", 0) * r.get("h", 0)))
json.dump(rows, open(os.path.join(out, "_textures.json"), "w"), indent=1)
for r in rows[:60]:
    print(r)

Step 4 (later, after the new art is approved): swap the frame without Unity

Two options, picked at design time:

A) Transparent frame: Unity draws nothing, and a sleek HTML/SVG overlay is drawn by the web page on top.
Easiest to iterate on.
Recommended.
B) New frame image: the new art (Nano Banana via the generate-image Supabase EF) is baked into the same texture slot.
Save as swap_texture.py and run python swap_texture.py <in.data> <out.data> <path_id> <new.png|transparent>:

import sys, UnityPy
from PIL import Image
src, dst, pid, new = sys.argv[1], sys.argv[2], int(sys.argv[3]), sys.argv[4]
env = UnityPy.load(src)
for obj in env.objects:
    if obj.path_id == pid and obj.type.name == "Texture2D":
        d = obj.read()
        w, h = d.image.size
        img = Image.new("RGBA", (w, h), (0, 0, 0, 0)) if new == "transparent" else Image.open(new).convert("RGBA").resize((w, h))
        d.image = img
        d.save()
        print("replaced", pid, d.m_Name, w, h)
open(dst, "wb").write(env.file.save())
