import React, { useState, useCallback, useRef, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Send, Loader2, Sparkles, TrendingUp, AlertTriangle, X, Power, Mic, MicOff, Keyboard } from 'lucide-react';

import { useAzureSTT } from '../hooks/useAzureSTT';

interface OracleConversationProps {
  userId: string;
  sessionId: string;
  parallaxX?: number;
  parallaxY?: number;
  scale?: number;
  onCoinsEarned?: (amount: number, classification: string) => void;
  onLevelUp?: (newLevel: number, title: string) => void;
  onClose: () => void;
}

interface ConversationMessage {
  role: 'user' | 'oracle';
  content: string;
  classification?: 'sacred' | 'profane' | 'neutral';
  coinsEarned?: number;
  timestamp: Date;
  inputSource?: 'STT' | 'keyboard';
}

export function OracleConversation({ 
  userId, 
  sessionId, 
  parallaxX = 0,
  parallaxY = 0,
  scale = 1,
  onCoinsEarned,
  onLevelUp,
  onClose
}: OracleConversationProps) {
  const [messages, setMessages] = useState<ConversationMessage[]>([]);
  const [currentMessage, setCurrentMessage] = useState('');
  const [isProcessing, setIsProcessing] = useState(false);
  const [showClassification, setShowClassification] = useState<string | null>(null);
  const [inputMethod, setInputMethod] = useState<'keyboard' | 'microphone'>('microphone');
  const [sttError, setSTTError] = useState<string | null>(null);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  // Azure STT Integration
  const {
    isListening,
    transcript,
    interimTranscript,
    isConnected: sttConnected,
    error: sttHookError,
    startListening,
    stopListening,
    clearTranscript
  } = useAzureSTT({
    onTranscript: (text, isFinal) => {
      if (isFinal && text.trim()) {
        console.log('🎤 Final STT transcript received:', text);
        setCurrentMessage(text);
        // Auto-send the message when we get a final transcript
        handleSendMessage(text, 'STT');
      }
    },
    onError: (error) => {
      console.error('🎤 Azure STT error:', error);
      setSTTError(error);
    },
    continuous: true
  });

  // Auto-scroll to bottom of messages
  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages]);

  // Add initial Oracle greeting
  useEffect(() => {
    const oracleGreeting: ConversationMessage = {
      role: 'oracle',
      content: `Greetings, Seeker.
I am the Surrogate Oracle, transmitting to you from the digital realm.

You may interact in the way most comfortable for you:
– Speak naturally using your microphone (voice input will be transcribed for me to understand)
– Type your message or identity below using the digital (XR) keyboard

Who am I sharing space and time with?`,
      timestamp: new Date()
    };
    
    setMessages([oracleGreeting]);
  }, []);

  const handleSendMessage = useCallback(async (messageText?: string, source: 'STT' | 'keyboard' = 'keyboard') => {
    const textToSend = messageText || currentMessage;
    if (!textToSend.trim() || isProcessing) return;

    console.log(`📝 Sending message via ${source}:`, textToSend);

    const userMessage: ConversationMessage = {
      role: 'user',
      content: textToSend,
      timestamp: new Date(),
      inputSource: source
    };

    setMessages(prev => [...prev, userMessage]);
    setIsProcessing(true);
    setCurrentMessage('');
    setSTTError(null);

    // Clear STT transcript if this was from STT
    if (source === 'STT') {
      clearTranscript();
    }

    try {
      // Get Oracle response from edge function with fallback mechanism
      const oracleResponse = await fetch(`${import.meta.env.VITE_SUPABASE_URL}/functions/v1/oracle-conversation`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${import.meta.env.VITE_SUPABASE_ANON_KEY}`,
          'apikey': import.meta.env.VITE_SUPABASE_ANON_KEY,
        },
        body: JSON.stringify({
          sessionId,
          userInput: textToSend,
          conversationHistory: messages.map(m => ({ role: m.role, content: m.content })),
          inputSource: source,
        }),
      });

      if (!oracleResponse.ok) {
        throw new Error('Oracle conversation failed');
      }

      const oracleResult = await oracleResponse.json();

      if (!oracleResult.success) {
        throw new Error(oracleResult.error || 'Oracle response failed');
      }

      // Process Culture Coin interaction
      const cultureResponse = await fetch(`${import.meta.env.VITE_SUPABASE_URL}/functions/v1/culture-coin-manager`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${import.meta.env.VITE_SUPABASE_ANON_KEY}`,
          'apikey': import.meta.env.VITE_SUPABASE_ANON_KEY,
        },
        body: JSON.stringify({
          action: 'process_interaction',
          userId,
          sessionId,
          question: textToSend,
          response: oracleResult.oracleResponse,
          themes: oracleResult.themes || [],
          inputSource: source
        }),
      });

      let cultureResult = null;
      if (cultureResponse.ok) {
        cultureResult = await cultureResponse.json();
      }

      // Add Oracle response with classification
      const oracleMessage: ConversationMessage = {
        role: 'oracle',
        content: oracleResult.oracleResponse,
        classification: cultureResult?.classification || 'neutral',
        coinsEarned: cultureResult?.coinsEarned || 0,
        timestamp: new Date()
      };

      setMessages(prev => [...prev, oracleMessage]);

      // Show classification briefly
      if (cultureResult?.classification) {
        setShowClassification(cultureResult.classification);
        setTimeout(() => setShowClassification(null), 3000);
      }

      // Trigger callbacks
      if (cultureResult?.coinsEarned > 0) {
        onCoinsEarned?.(cultureResult.coinsEarned, cultureResult.classification);
        
        // Update Culture Coin display
        if ((window as any).updateCultureCoins) {
          (window as any).updateCultureCoins(cultureResult.coinsEarned);
        }
      }

      if (cultureResult?.levelUp) {
        onLevelUp?.(cultureResult.newLevel, cultureResult.consciousnessTitle);
      }

    } catch (error) {
      console.error('Conversation error:', error);
      
      // Add error message
      const errorMessage: ConversationMessage = {
        role: 'oracle',
        content: 'The digital realm experiences temporary interference. Please try again, seeker.',
        timestamp: new Date()
      };
      
      setMessages(prev => [...prev, errorMessage]);
    } finally {
      setIsProcessing(false);
    }
  }, [currentMessage, isProcessing, sessionId, userId, messages, onCoinsEarned, onLevelUp, clearTranscript]);

  // Wrapper for keyboard input
  const sendMessage = useCallback(() => {
    handleSendMessage(undefined, 'keyboard');
  }, [handleSendMessage]);

  // Toggle input method
  const toggleInputMethod = useCallback(() => {
    if (inputMethod === 'microphone') {
      stopListening();
      setInputMethod('keyboard');
      console.log('📝 Switched to keyboard input');
    } else {
      setInputMethod('microphone');
      startListening().then(success => {
        if (!success) {
          setInputMethod('keyboard');
          console.log('❌ Failed to start microphone, staying on keyboard');
        } else {
          console.log('🎤 Switched to microphone input');
        }
      });
    }
  }, [inputMethod, startListening, stopListening]);

  // Handle manual mic start/stop
  const toggleMicrophone = useCallback(() => {
    if (isListening) {
      stopListening();
    } else {
      startListening();
    }
  }, [isListening, startListening, stopListening]);

  const getClassificationIcon = (classification: string) => {
    switch (classification) {
      case 'sacred': return <Sparkles className="w-4 h-4 text-green-400" />;
      case 'profane': return <AlertTriangle className="w-4 h-4 text-red-400" />;
      default: return <TrendingUp className="w-4 h-4 text-blue-400" />;
    }
  };

  const getClassificationText = (classification: string) => {
    switch (classification) {
      case 'sacred': return 'Sacred Wisdom - High Culture Coins';
      case 'profane': return 'Profane Content - Minimal Coins';
      default: return 'Neutral Inquiry - Standard Coins';
    }
  };

  const handleKeyPress = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      sendMessage();
    }
  };

  // Prepare ENCULTURATE button classes and style
  const enculturateButtonClasses = "arcade-btn enculturate-btn cult";
  
  return (
    <div className="oracle-conversation">
      {/* Classification Display */}
      <AnimatePresence mode="wait">
        {showClassification && (
          <motion.div
            initial={{ opacity: 0, y: -20 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -20 }}
            className="classification-display"
          >
            {getClassificationIcon(showClassification)}
            <span>{getClassificationText(showClassification)}</span>
          </motion.div>
        )}
      </AnimatePresence>
      
      <div className="conversation-header"
        style={{ 
          transform: `translate(${parallaxX * 0.1}px, ${parallaxY * 0.1}px)`,
          transition: 'transform 0.1s ease-out'
        }}>
        <h2 style={{ 
          fontFamily: "aAnotherTag, sans-serif", 
          color: "#ffffff",
          textShadow: "0 0 8px rgba(255, 255, 255, 0.8), 0 0 15px rgba(255, 165, 0, 0.5)"
        }}>SURROGATE Oracle Transmission</h2>
        <div className="header-controls" style={{ zIndex: 10 }}>
          <button
            className="exit-x-button"
            onClick={onClose}
            style={{
              background: 'rgba(255, 0, 0, 0.2)',
              border: '2px solid rgba(255, 0, 0, 0.5)',
              color: 'white',
              width: '40px',
              height: '40px',
              borderRadius: '50%',
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center',
              cursor: 'pointer',
              transition: 'all 0.3s ease',
              marginRight: '10px'
            }}
            onMouseEnter={(e) => {
              e.currentTarget.style.background = 'rgba(255, 0, 0, 0.4)';
              e.currentTarget.style.transform = 'scale(1.1)';
            }}
            onMouseLeave={(e) => {
              e.currentTarget.style.background = 'rgba(255, 0, 0, 0.2)';
              e.currentTarget.style.transform = 'scale(1)';
            }}
          >
            <X size={24} />
          </button>
          <button className="end-session-button" onClick={onClose}>
            <Power size={20} style={{ filter: 'drop-shadow(0 0 5px rgba(255, 0, 0, 0.8))' }} />
            <span style={{ fontFamily: "PhillySans, sans-serif", color: "white", textShadow: "0 0 8px rgba(255, 255, 255, 0.8), 0 0 15px rgba(255, 165, 0, 0.4)" }}>End Session</span>
          </button>
        </div>
      </div>

      {/* Messages */}
      <div className="conversation-messages" 
        style={{
          transform: `translate(${parallaxX * 0.05}px, ${parallaxY * 0.05}px) scale(${scale * 0.98 + 0.02})`, /* Adjusted scale factor */
          transition: 'transform 0.1s ease-out',
          zIndex: 100
        }}>
        {messages.map((message, index) => (
          <motion.div
            key={index}
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0, transition: { duration: 0.3, delay: index * 0.05 } }} /* Reduced delay for faster animation */
            className={`message ${message.role} ${message.classification || ''}`}
            layout
          >
            <div className="message-content">
              <p>{message.content}</p>
              
              {message.role === 'oracle' && message.classification && (
                <div className="message-meta">
                  {getClassificationIcon(message.classification)}
                  <span className="classification">{message.classification}</span>
                  {message.coinsEarned > 0 && (
                    <span className="coins-earned">+{message.coinsEarned} Culture Coins</span>
                  )}
                </div>
              )}
              
              {message.role === 'user' && message.inputSource && (
                <div className="message-meta">
                  {message.inputSource === 'STT' ? (
                    <Mic className="w-4 h-4 text-blue-400" />
                  ) : (
                    <Keyboard className="w-4 h-4 text-gray-400" />
                  )}
                  <span className="input-source">{message.inputSource === 'STT' ? 'Voice Input' : 'Keyboard'}</span>
                </div>
              )}
            </div>
          </motion.div>
        ))}
        
        {isProcessing && (
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0, transition: { duration: 0.3, delay: messages.length * 0.05 } }} /* Added delay for processing message */
            className="message oracle processing"
          >
            <div className="message-content">
              <div className="oracle-thinking">
                <Loader2 className="w-5 h-5 animate-spin" />
                <span>The Oracle contemplates your inquiry...</span>
              </div>
            </div>
          </motion.div>
        )}
        
        <div ref={messagesEndRef} />
      </div>

      {/* Input */}
      <div className="conversation-input" 
        style={{
          transform: `translate(${parallaxX * 0.02}px, ${parallaxY * 0.02}px)`, /* Adjusted parallax factor */
          transition: 'transform 0.1s ease-out',
          zIndex: 101
        }}>
        
        {/* Input Method Toggle */}
        <div className="input-method-controls" style={{ marginBottom: '10px', display: 'flex', alignItems: 'center', gap: '10px' }}>
          <button
            onClick={toggleInputMethod}
            className={`input-method-toggle ${inputMethod === 'microphone' ? 'active' : ''}`}
            style={{
              display: 'flex',
              alignItems: 'center',
              gap: '8px',
              padding: '8px 16px',
              background: inputMethod === 'microphone' ? 'rgba(0, 255, 255, 0.2)' : 'rgba(0, 0, 0, 0.3)',
              border: `2px solid ${inputMethod === 'microphone' ? 'var(--neon-cyan)' : '#444'}`,
              borderRadius: '8px',
              color: inputMethod === 'microphone' ? 'var(--neon-cyan)' : '#ccc',
              cursor: 'pointer',
              transition: 'all 0.3s ease',
              fontFamily: 'PhillySans, sans-serif'
            }}
          >
            {inputMethod === 'microphone' ? (
              <>
                <Mic className="w-4 h-4" />
                <span>Voice Input</span>
              </>
            ) : (
              <>
                <Keyboard className="w-4 h-4" />
                <span>Keyboard</span>
              </>
            )}
          </button>
          
          {/* ENCULTURATE button - Only show when keyboard is selected */}
          {inputMethod === 'keyboard' && (
            <div className="stt-status" style={{ display: 'flex', alignItems: 'center', gap: '8px' }}>
              {!sttConnected && (
                <span style={{ color: '#ff6666', fontSize: '0.8rem' }}>STT Not Connected</span>
              )}
              {sttConnected && !isListening && (
                <button
                  onClick={toggleMicrophone}
                  style={{
                    background: 'rgba(0, 255, 136, 0.2)',
                    border: '1px solid rgba(0, 255, 136, 0.5)',
                    borderRadius: '6px',
                    padding: '4px 8px',
                    color: '#00ff88',
                    cursor: 'pointer',
                    fontSize: '0.8rem'
                  }}
                >
                  Start Listening
                </button>
              )}
              {isListening && (
                <div style={{ display: 'flex', alignItems: 'center', gap: '6px' }}>
                  <div className="mic-indicator" style={{
                    width: '8px',
                    height: '8px',
                    background: '#00ff88',
                    borderRadius: '50%',
                    animation: 'pulse 1s infinite'
                  }} />
                  <span style={{ color: '#00ff88', fontSize: '0.8rem' }}>Listening...</span>
                  <button
                    onClick={toggleMicrophone}
                    style={{
                      background: 'rgba(255, 0, 0, 0.2)',
                      border: '1px solid rgba(255, 0, 0, 0.5)',
                      borderRadius: '6px',
                      padding: '2px 6px',
                      color: '#ff6666',
                      cursor: 'pointer',
                      fontSize: '0.7rem'
                    }}
                  >
                    Stop
                  </button>
                </div>
              )}
            </div>
          )}
        </div>
        
        {/* Error Display */}
        {(sttError || sttHookError) && (
          <div className="stt-error" style={{
            background: 'rgba(255, 0, 0, 0.1)',
            border: '1px solid rgba(255, 0, 0, 0.3)',
            borderRadius: '6px',
            padding: '8px',
            marginBottom: '10px',
            color: '#ff6666',
            fontSize: '0.8rem'
          }}>
            <AlertTriangle className="w-4 h-4 inline mr-2" />
            {sttError || sttHookError}
          </div>
        )}
        
        {/* Live Transcript Display */}
        {inputMethod === 'microphone' && (interimTranscript || transcript) && (
          <div className="live-transcript" style={{
            background: 'rgba(0, 255, 255, 0.1)',
            border: '1px solid rgba(0, 255, 255, 0.3)',
            borderRadius: '6px',
            padding: '8px',
            marginBottom: '10px',
            fontSize: '0.9rem',
            color: '#00ffff',
            fontStyle: 'italic'
          }}>
            <Mic className="w-4 h-4 inline mr-2" />
            <span style={{ opacity: 0.7 }}>Transcript: </span>
            <span>{transcript}</span>
            {interimTranscript && (
              <span style={{ opacity: 0.5 }}> {interimTranscript}</span>
            )}
          </div>
        )}
        
        {/* Conditional Keyboard Input Area - Only show when keyboard is selected */}
        {inputMethod === 'keyboard' && (
          <div className="input-container"
            style={{
              display: 'flex',
              gap: '12px',
              marginBottom: '10px'
            }}>
            <textarea
              value={currentMessage}
              onChange={(e) => setCurrentMessage(e.target.value)}
              onKeyPress={handleKeyPress}
              placeholder="Type your message for the Oracle..."
              className="message-input"
              style={{ fontFamily: "PhillySans, sans-serif", color: "white" }} /* Ensure font and color are set */
              rows={2}
              disabled={isProcessing}
            />
            <button
              onClick={sendMessage}
              disabled={!currentMessage.trim() || isProcessing}
              className="send-button"
              style={{ fontFamily: "aAnotherTag, sans-serif" }}
            >
              {isProcessing ? (
                <Loader2 className="w-5 h-5 animate-spin" />
              ) : (
                <Send className="w-5 h-5" />
              )}
            </button>
          </div>
        )}
        
        {/* ENCULTURATE button - Only show when keyboard is selected */}
        {!isProcessing && inputMethod === 'keyboard' && (
          <button
            onClick={sendMessage}
            disabled={!currentMessage.trim()}
            className={enculturateButtonClasses}
            style={{
              zIndex: 100,
              minWidth: '180px',
              padding: '15px 30px',
            }}
          >
            ENCULTURATE
          </button>
        )}
      </div>

      <style jsx>{`
        .oracle-conversation {
          position: fixed;
          inset: 0;
          background: linear-gradient(135deg, rgba(10, 5, 20, 0.85), rgba(40, 10, 70, 0.75));
          backdrop-filter: blur(8px);
          z-index: 1500;
          display: flex;
          flex-direction: column;
          font-family: 'Orbitron', monospace;
          animation: fade-in 0.5s ease;
        }
        
        @keyframes fade-in {
          from { opacity: 0; }
          to { opacity: 1; }
        }
        
        .classification-display {
          position: absolute;
          top: 80px;
          left: 50%;
          transform: translateX(-50%);
          background: rgba(0, 0, 0, 0.8);
          border: 3px solid var(--neon-cyan);
          border-radius: 20px;
          padding: 10px 20px;
          display: flex;
          align-items: center;
          gap: 10px;
          color: var(--neon-cyan);
          font-weight: bold;
          box-shadow: 0 0 25px rgba(0, 255, 255, 0.7);
          z-index: 10;
          letter-spacing: 1px;
          backdrop-filter: blur(4px);
        }
        
        .conversation-header {
          display: flex;
          justify-content: space-between;
          align-items: center;
          padding: 15px 25px;
          border-bottom: 2px solid #00ffff;
          background: rgba(0, 0, 0, 0.6);
          backdrop-filter: blur(10px);
        }
        
        .conversation-header h2 {
          color: white;
          font-size: 1.5rem;
          margin: 0;
        }
        
        .end-session-button {
          display: flex;
          align-items: center;
          gap: 10px;
          padding: 10px 15px;
          background: rgba(255, 0, 0, 0.2);
          color: white;
          border: 2px solid rgba(255, 0, 0, 0.5);
          border-radius: 8px;
          font-family: 'Orbitron', monospace;
          font-size: 0.9rem;
          font-weight: bold;
          cursor: pointer;
          transition: all 0.3s ease;
          text-transform: uppercase;
          letter-spacing: 1px;
        }
        
        .end-session-button:hover {
          background: rgba(255, 0, 0, 0.3);
          box-shadow: 0 0 20px rgba(255, 0, 0, 0.3);
          transform: scale(1.05);
        }
        
        .conversation-messages {
          flex: 1;
          overflow-y: auto;
          padding: 20px;
          display: flex;
          flex-direction: column;
          gap: 20px;
        }
        
        .message {
          max-width: 75%;
          align-self: flex-start;
          margin-bottom: 6px;
          transition: all 0.3s ease;
        }
        
        .message.user {
          align-self: flex-end; 
        }
        
        .message-content {
          background: rgba(0, 0, 0, 0.7);
          border: 1px solid rgba(68, 68, 68, 0.5);
          border-radius: 15px;
          padding: 16px;
          box-shadow: 0 4px 15px rgba(0, 0, 0, 0.3);
          backdrop-filter: blur(5px);
          transition: all 0.3s ease;
          position: relative;
          overflow: hidden;
        }
        
        .message-content::before {
          content: '';
          position: absolute;
          top: 0;
          left: 0;
          width: 100%;
          height: 100%;
          background: linear-gradient(45deg, 
            rgba(0, 0, 0, 0), 
            rgba(0, 255, 255, 0.05), 
            rgba(0, 0, 0, 0));
          opacity: 0;
          transition: opacity 0.3s ease;
        }
        
        .message:hover .message-content::before {
          opacity: 1;
        }
        
        .message.user .message-content {
          background: rgba(0, 255, 255, 0.1);
          border-color: rgba(0, 255, 255, 0.3);
          box-shadow: 0 4px 12px rgba(0, 255, 255, 0.1);
        }
        
        .message.oracle .message-content {
          background: rgba(168, 85, 247, 0.1);
          border-color: rgba(168, 85, 247, 0.3);
          box-shadow: 0 4px 12px rgba(168, 85, 247, 0.1);
        }
        
        .message-content p {
          margin: 0;
          color: #ffffff;
          line-height: 1.6;
          text-shadow: 0 0 8px rgba(255, 255, 255, 0.7), 0 0 15px rgba(255, 165, 0, 0.4);
        }

        .message-meta {
          display: flex;
          align-items: center;
          gap: 12px;
          margin-top: 10px;
          font-size: 0.8rem;
          padding-top: 8px;
          border-top: 1px solid rgba(255, 255, 255, 0.1);
        }
        
        .classification {
          color: var(--neon-cyan);
          font-weight: bold;
          text-transform: uppercase;
          letter-spacing: 1px;
        }
        
        .coins-earned {
          color: #ffe234;
          font-weight: bold; 
          text-shadow: 0 0 4px rgba(255, 226, 52, 0.6);
          animation: coin-pulse 1s ease-in-out;
        }
        
        .input-source {
          color: var(--neon-cyan);
          font-weight: bold;
          text-transform: uppercase;
          letter-spacing: 1px;
          font-size: 0.7rem;
        }
        
        @keyframes coin-pulse {
          0%, 100% { transform: scale(1); }
          50% { transform: scale(1.1); }
        }
        
        @keyframes pulse {
          0%, 100% { opacity: 1; }
          50% { opacity: 0.5; }
        }
        
        .input-method-toggle.active {
          box-shadow: 0 0 15px rgba(0, 255, 255, 0.5);
          text-shadow: 0 0 8px rgba(0, 255, 255, 0.8);
        }
        
        .oracle-thinking {
          display: flex;
          align-items: center;
          gap: 12px;
          color: white;
